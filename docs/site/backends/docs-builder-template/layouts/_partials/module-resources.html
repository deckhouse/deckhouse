{{/* Renders a module resources (CRDs or configuration OpenAPI specification). Context: .name — module name .channel — module channel .data — module data (usually, it is hugo.Data.modules. structure) .type — resource to render. {{ T "version" | humanize }}: {{ $version }}

{{ partial "openapi/format-description" (or $langSchema.description $schema.description) }}
{{- if or (isset $schema "x-doc-examples") (isset $schema "x-doc-example") (isset $schema "example") (isset $schema "x-examples") }} {{- template "format-example" (dict "attributes" $schema "langData" $langSchema "linkAnchor" $resourceName "name" "") }} {{- end }}
    {{- range $name, $property := $schema.properties }} {{- $propertyLangData := index $langSchema.properties $name }} {{- partial "openapi/format-schema" (dict "name" $name "data" $property "langData" $propertyLangData "parent" $schema "resourceName" $resourceName "required" $schema.required) }} {{- end }}
{{- end }} {{- end }} {{- else if and (eq .type "configuration") $moduleData.openapi }} {{- $cfgData := index $moduleData.openapi "config-values" }} {{- $cfgConversions := index $moduleData.openapi "conversions" }} {{ if $cfgData }} {{- $langData := index $moduleData.openapi (printf "doc-%s-config-values" $lang ) }} {{- partial "openapi/format-configuration" ( dict "name" $moduleName "channel" $moduleChannel "data" $cfgData "conversions" $cfgConversions "langData" $langData "pageParams" $pageParams ) }} {{ end }} {{- end }}
