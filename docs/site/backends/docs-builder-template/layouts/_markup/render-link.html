{{- /* Markdown link render hook. Delegates to the shared helper so that links from Markdown and links emitted by HTML templates go through the same logic. Per-link opt-out via the Markdown link title: [text](url "keep-link"). Hugo/goldmark does not accept inline Kramdown-style attributes on links, so the title attribute is used as the marker (equivalent to Jekyll's data-keep-link). The "keep-link" marker is consumed here and not emitted as a title attribute. See layouts/_partials/helpers/rewrite-href.html for the rewrite logic and config. */ -}} {{- $keep := eq .Title "keep-link" -}} {{- $attrs := "" -}} {{- if and (not $keep) .Title -}} {{- $attrs = printf " title=\"%s\"" .Title -}} {{- end -}} {{- /* On a site of one module version (docs-builder RenderVersion writes data/version_site.json), keep links to the same module inside that version: /modules//x.html -> //modules///x.html. Links that already name a channel or a version and links to other modules are left untouched. */ -}} {{- $dest := .Destination -}} {{- with hugo.Data.version_site -}} {{- $own := printf `^(/(ru|en))?/modules/%s/` .module -}} {{- $pinned := printf `^(/(ru|en))?/modules/%s/(alpha|beta|early-access|stable|rock-solid|latest|v[0-9][^/]*|%s)/` .module (replaceRE `\.` `\.` .version) -}} {{- if and (findRE $own $dest 1) (not (findRE $pinned $dest 1)) -}} {{- $dest = replaceRE $own (printf "/%s/modules/%s/%s/" $.Page.Language.Lang .module .version) $dest 1 -}} {{- end -}} {{- end -}} {{- partial "helpers/rewrite-link" (dict "href" $dest "inner" .Text "attrs" $attrs "keep" $keep) -}}
